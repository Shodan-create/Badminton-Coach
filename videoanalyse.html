const videoElement = document.getElementById('inputVideo'); const canvasElement = document.getElementById('outputCanvas'); const canvasCtx = canvasElement.getContext('2d'); const fileInput = document.getElementById('videoUpload'); const rhythmScore = document.getElementById('rhythmScore'); const cadenceValue = document.getElementById('cadenceValue'); const flowStatus = document.getElementById('flowStatus'); let historySpeeds = []; let stepTimestamps = []; // Speichert Zeitpunkte von echten Schritten let prevFootX = null; let prevFootY = null; let isMovingUp = false; const pose = new Pose({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}` }); pose.setOptions({ modelComplexity: 1, smoothLandmarks: true, minDetectionConfidence: 0.3, minTrackingConfidence: 0.3 }); pose.onResults((results) => { canvasElement.width = videoElement.clientWidth; canvasElement.height = videoElement.clientHeight; canvasCtx.save(); canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height); if (results.poseLandmarks) { const landmarks = results.poseLandmarks; drawConnectors(canvasCtx, landmarks, POSE_CONNECTIONS, { color: '#00f2fe', lineWidth: 2 }); drawLandmarks(canvasCtx, landmarks, { color: '#ffe600', lineWidth: 1, radius: 4 }); const leftAnkle = landmarks[27]; const rightAnkle = landmarks[28]; if (leftAnkle && rightAnkle) { const currentFootX = (leftAnkle.x + rightAnkle.x) / 2; const currentFootY = (leftAnkle.y + rightAnkle.y) / 2; if (prevFootX !== null) { const dx = (currentFootX - prevFootX) * canvasElement.width; const dy = (currentFootY - prevFootY) * canvasElement.height; const stepSpeed = Math.sqrt(dx * dx + dy * dy); historySpeeds.push(stepSpeed); if (historySpeeds.length > 45) historySpeeds.shift(); // Schaut sich ca. 1.5 Sekunden an // 1. SCHRITTFREQUENZ / CADENCE (PEAK-ERKENNUNG) // Ein Schritt gilt als getätigt, wenn die Bewegung nach einer Beschleunigung wieder abfällt if (stepSpeed > 2.5 && !isMovingUp) { isMovingUp = true; const now = videoElement.currentTime; stepTimestamps.push(now); } else if (stepSpeed < 1.5) { isMovingUp = false; } // Nur Schritte der letzten 5 Sekunden im Video berücksichtigen const now = videoElement.currentTime; stepTimestamps = stepTimestamps.filter(t => (now - t) <= 5); // Hochrechnen auf Schritte pro Minute (BPM / Cadence) const stepsPerMinute = Math.round((stepTimestamps.length / 5) * 60); cadenceValue.innerText = `${stepsPerMinute} Schritte/Min`; // 2. STRENGERE & REALISTISCHERE FLÜSSIGKEITS-BERECHNUNG const avgSpeed = historySpeeds.reduce((a, b) => a + b, 0) / historySpeeds.length; // Varianz / Ruckel-Faktor berechnen let varianceSum = 0; for (let i = 0; i < historySpeeds.length; i++) { varianceSum += Math.abs(historySpeeds[i] - avgSpeed); } const meanJerk = varianceSum / historySpeeds.length; // Neuer strenger Algorithmus: Ruckler & harte Stops drücken den Score spürbar! // Ein "Pro Flow" liegt meist zwischen 75% und 92%. Nur perfektes Gleiten erreicht >90%. let realisticRhythm = Math.round(100 - (meanJerk * 18)); if (realisticRhythm > 98) realisticRhythm = 95; // Hard-Cap für Realismus if (realisticRhythm < 10) realisticRhythm = 10; rhythmScore.innerText = `${realisticRhythm}%`; // 3. HARDCORE-EINSTUFUNG DER STATUS-TEXTE if (realisticRhythm >= 82) { rhythmScore.className = 'metric-value status-good'; flowStatus.innerText = '✨ Flüssiger Pro-Flow'; flowStatus.className = 'metric-value status-good'; } else if (realisticRhythm >= 60) { rhythmScore.className = 'metric-value'; flowStatus.innerText = '⚖️ Solider Rhythmus (Tempo-Wechsel)'; flowStatus.className = 'metric-value'; } else { rhythmScore.className = 'metric-value status-warn'; flowStatus.innerText = '⚠️ Stopp-and-Go / Ruckelig'; flowStatus.className = 'metric-value status-warn'; } // 4. RHYTHMUS-GRAPH canvasCtx.beginPath(); canvasCtx.strokeStyle = realisticRhythm >= 82 ? '#3fb950' : '#d29922'; canvasCtx.lineWidth = 2; const graphWidth = 160; const graphHeight = 45; const startX = canvasElement.width - graphWidth - 15; const startY = 15; canvasCtx.fillStyle = 'rgba(13, 17, 23, 0.85)'; canvasCtx.fillRect(startX - 8, startY - 8, graphWidth + 16, graphHeight + 16); historySpeeds.forEach((spd, idx) => { const x = startX + (idx * (graphWidth / 45)); const y = startY + graphHeight - Math.min(graphHeight, spd * 3); if (idx === 0) canvasCtx.moveTo(x, y); else canvasCtx.lineTo(x, y); }); canvasCtx.stroke(); } prevFootX = currentFootX; prevFootY = currentFootY; } } }); async function processFrame() { if (!videoElement.paused && !videoElement.ended) { await pose.send({ image: videoElement }); requestAnimationFrame(processFrame); } } videoElement.addEventListener('play', () => processFrame()); fileInput.addEventListener('change', (e) => { const file = e.target.files[0]; if (file) { videoElement.src = URL.createObjectURL(file); videoElement.load(); historySpeeds = []; stepTimestamps = []; prevFootX = null; } });
